je_lou_frontier_migration = {
	name = "je_lou_frontier_migration_name"
	desc = "je_lou_frontier_migration_desc"

	visible = { always = yes }
	

	ai_chance = {
		base = 10
	}

	possible = {
		NOT = { has_modifier = lou_early_migrant_cost }
		# Note that this closely reflects triggers in the event. Do not change one without changing both
		any_country = {
			country_rank <= rank_value:major_power
			custom_tooltip = {
				text = is_in_europe_tt
				country_is_in_europe = yes
			}
			custom_tooltip = {
				text = same_state_religion
				religion = root.religion
			}
			custom_tooltip = {
				text = average_sol_below_root_tt
				average_sol <= root.average_sol
			}
			any_primary_culture = {
				culture_can_have_mass_migration_to_country = ROOT
			}
			any_scope_state = {
				is_incorporated = yes
				custom_tooltip = {
					text = is_in_europe_tt
					state_is_in_europe = yes
				}
				count >= 3
			}
		}
		any_scope_state = {
			trigger_if = {
			}
			state_is_eligible_as_mass_migration_target = yes
		}
		custom_tooltip = {
		text = lou_immi_tolerance_tt
		OR = {
			has_law = law_type:law_racial_segregation
			has_law = law_type:law_cultural_exclusion
			has_law = law_type:law_multicultural
		}
		NOR = {
			has_law = law_type:law_closed_borders
			has_law = law_type:law_isolationism
		}
	}
	}
	effect = {
	trigger_event = { id = lou_frontier.1 popup = yes }
	 }
	}
